 / med_price) <= 0.02)
        spread = max((v[0] for v in votes), default=0.0) - min((v[0] for v in votes), default=0.0)
        spread_pct = (spread / med_price) * 100 if med_price else 0.0
        prices[tk] = {
            "price": med_price, "change_pct": change_pct,
            "volume_24h": vol, "source": ref[3],
            "spread_pct": spread_pct, "agree": agree,
        }

    for name in METAL_SYMBOLS:
        prices[name] = metals.get(name) or {"price": 0.0, "change_pct": 0.0,
                                            "source": "none"}

    return prices


def verify_post_data(prices: dict[str, dict]) -> tuple[bool, str]:
    """Перед отправкой — все основные тикеры имеют цену > 0 и согласованы."""
    issues: list[str] = []
    for tk in CRYPTO_TICKERS:
        row = prices.get(tk, {})
        px = row.get("price", 0.0)
        if px <= 0:
            issues.append(f"{tk}=0")
            continue
        if row.get("agree", 0) < 2 and len(PRICE_SOURCES) >= 3:
            issues.append(f"{tk} low agree={row.get('agree')}")
        if row.get("spread_pct", 0.0) > 5.0:
            issues.append(f"{tk} spread={row.get('spread_pct'):.1f}%")
    for name in METAL_SYMBOLS:
        row = prices.get(name, {})
        if row.get("price", 0.0) <= 0:
            issues.append(f"{name}=0")
    if issues:
        return False, "; ".join(issues)
    return True, "ok"
