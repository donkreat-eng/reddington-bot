
            if d:

def fetch_kraken(symbols: dict[str, str]) -> dict[str, dict]:
    """Kraken public Ticker. Переводит тикер Binance -> пара Kraken."""
    KRAKEN_PAIR = {"BTC": "XBT", "ETH": "ETH", "BNB": "BNB",
                   "SOL": "SOL", "XRP": "XRP"}
    out: dict[str, dict] = {}
    base = "https://api.kraken.com/0/public/Ticker"
    for tk, sym in symbols.items():
        kraken_sym = f"{KRAKEN_PAIR.get(tk, tk)}USD"
        try:
            data = _http_json(base, params={"pair": kraken_sym})
            res = data.get("result", {})
            if not res:
                continue
            payload = next(iter(res.values()))
            close = float(payload["c"][0])
            open_ = float(payload["o"])
            change_pct = ((close - open_) / open_) * 100 if open_ else 0.0
            vol = float(payload["v"][1])  # today volume
            out[tk] = {"price": close, "change_pct": change_pct,
                       "volume_24h": vol * close, "source": "kraken"}
        except Exception as e:  # noqa: BLE001
            log("market_overview", f"kraken {tk}: {e}")
        _throttle("kraken")
    return out
