    lines.append("")
    for name in METAL_SYMBOLS:
        row = prices[name]
        arrow = "🟢" if row["change_pct"] >= 0 else "🔴"
        sign = "+" if row["change_pct"] >= 0 else ""
        lines.append(
            f"{arrow} <b>{name}</b>  <code>${fmt_price(row['price'])}</code>  "
            f"{sign}{row['change_pct']:.2f}%"
        )
    if fng:
        lines.append("")
        lines.append(f"😨 <b>F&G</b>: {fng['value']} — {fng['classification']}")
    return "\n".join(lines)


# --- Сборка позиционного слоя ---------------------------------------------

def build_position_layer(prices: dict[str, dict]) -> str:
    """Один-два абзаца про контекст: BTC доминирование не считаем,
    опираемся на расхождения источников, волатильность и поведение метала.
    Без рекомендаций, без явных уровней — только описание режима."""
    notes: list[str] = []
    btc = prices.get("BTC", {})
    if btc.get("agree", 0) >= 3:
        notes.append("BTC согласовано по 3+ источникам — режим спокойный.")
    elif btc.get("agree", 0) == 2:
        notes.append("BTC: 2 из {} источников подтверждают медиану — возможен локальный перекос.".format(
            len(PRICE_SOURCES)))
    else:
        notes.append("⚠️ BTC: источники расходятся, цена неустойчива.")
    if btc.get("spread_pct", 0.0) > 1.0:
        notes.append(f"Спред BTC между биржами {btc['spread_pct']:.2f}% — ликвидность фрагментирована.")
    g = prices.get("Gold", {})
    s = prices.get("Silver", {})
    if g.get("price") and s.get("price"):
        ratio = s["price"] / g["price"] if g["price"] else 0
        notes.append(f"Gold/Silver ratio ≈ {ratio:.1f}.")
    if not notes:
        return ""
    return "🧭 <b>Контекст</b>:\n" + "\n".join(f"  • {n}" for n in notes)

def post_pair(caption: str, body: str) -> None:
    """POST в Telegram. Простой HTML parse_mode."""
    import urllib.parse
    import urllib.request
    text = f"<b>{caption}</b>\n\n{body}"
    data = urllib.parse.urlencode({
        "chat_id": TELEGRAM_CHAT,
        "text": text,
        "parse_mode": "HTML",
        "disable_web_page_preview": "true",
    }).encode("utf-8")
    req = urllib.request.Request(
        f"https://api.telegram.org/bot{TELEGRAM_TOKEN}/sendMessage",
        data=data,
    )
