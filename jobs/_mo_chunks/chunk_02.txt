n()
        except Exception as e:  # noqa: BLE001
            last = e
            time.sleep(2 ** attempt)
    raise RuntimeError(f"http failed after retries: {last}")


def _throttle(source: str) -> None:
    time.sleep(THROTTLE.get(source, 1.0))

def fetch_binance(symbols: dict[str, str]) -> dict[str, dict]:
    """/ticker/24hr для пар из BINANCE_SYMBOL. USDT-котировки."""
    out: dict[str, dict] = {}
    for tk, sym in symbols.items():
        url = f"https://api.binance.com/api/v3/ticker/24hr?symbol={sym}"
        try:
            data = _http_json(url)
            out[tk] = {
                "price": float(data["lastPrice"]),
                "change_pct": float(data["priceChangePercent"]),
                "volume_24h": float(data["quoteVolume"]),
                "source": "binance",
            }
        except Exception as e:  # noqa: BLE001
            log("market_overview", f"binance {tk}: {e}")
        _throttle("binance")
    return out
