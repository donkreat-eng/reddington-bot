
            if data and "price" in data:
                px = float(data["price"])
                # Coinbase не возвращает 24h change в /ticker; берём из /stats
                stats = _http_json(url.replace("/ticker", "/stats"))
                open_ = float(stats.get("open", px)) if stats else px
                change_pct = ((px - open_) / open_) * 100 if open_ else 0.0
                vol = float(stats.get("volume", 0.0)) if stats else 0.0
                out[tk] = {"price": px, "change_pct": change_pct,
                           "volume_24h": vol * px, "source": "coinbase"}
        except Exception as e:  # noqa: BLE001
            log("market_overview", f"coinbase {tk}: {e}")
        _throttle("coinbase")
    return out


def fetch_metal(symbol: str) -> dict | None:
    """Yahoo Finance quote по тикеру GC=F/SI=F."""
    url = f"https://query1.finance.yahoo.com/v7/finance/quote"
    try:
        data = _http_json(url, params={"symbols": symbol})
        res = (data.get("quoteResponse") or {}).get("result") or []
        if not res:
            return None
        q = res[0]
        price = q.get("regularMarketPrice")
        prev = q.get("regularMarketPreviousClose") or price
        if price is None:
            return None
        change_pct = ((price - prev) / prev) * 100 if prev else 0.0
        return {"price": float(price), "change_pct": float(change_pct),
                "source": "yahoo"}
    except Exception as e:  # noqa: BLE001
        log("market_overview", f"yahoo {symbol}: {e}")
        return None


def fetch_fng() -> dict | None:
    """Alternative.me Crypto Fear & Greed Index."""
    try:
        data = _http_json("https://api.alternative.me/fng/?limit=1")
        if not data or "data" not in data or not data["data"]:
            return None
        d = data["data"][0]
        return {"value": int(d["value"]), "classification": d["value_classification"]}
    except Exception as e:  # noqa: BLE001
        log("market_overview", f"fng: {e}")
        return None
