def _http_json(url: str, params: dict | None = None, timeout: int = FETCH_TIMEOUT) -> Any:
    """GET с ретраями; возвращает распарсенный JSON или поднимает."""
    headers = {"User-Agent": USER_AGENT, "Accept": "application/json"}
    last: Exception | None = None
    for attempt in range(3):
        try:
            r = requests.get(url, params=params or {}, headers=headers, timeout=timeout)
            if r.status_code == 429 or r.status_code >= 500:
                raise RuntimeError(f"upstream {r.status_code}")
            r.raise_for_status()
            return r.json()
        except Exception as e:  # noqa: BLE001
            last = e
            time.sleep(2 ** attempt)
    raise RuntimeError(f"http failed after retries: {last}")


def _throttle(source: str) -> float:
    return THROTTLE.get(source, FETCH_PER_REQUEST_SEC)
